Price calculator

Add rows, adjust quantities, and watch the total recompute from the expression sum('#list .amount') after every change. Press Escape in a field to reset it; prevent-default cancels the browser’s own revert first.

<div class="demo">
      <div class="row">
        <label>Qty
          <input id="qty" implements="modifiable dirtyable attributable prevent-default"
                 type="number" value="1" min="1" max="10"
                 on-input="#preview.set(this.value)"
                 on-dirty="this.setAttr({name: 'data-dirty', value: ''})"
                 on-clean="this.removeAttr('data-dirty')"
                 on-keydown(code:`Escape`)="this.reset(); #preview.set(#qty.value)">
        </label>
        <button on-click="#qty.set(#qty.value - 1); #preview.set(#qty.value)">−</button>
        <button on-click="#qty.set(#qty.value + 1); #preview.set(#qty.value)">+</button>
        <button on-click="#qty.set(#qty.value + 5); #preview.set(#qty.value)">+5</button>
        <button on-click="#qty.reset(); #preview.set(#qty.value)">Reset</button>
        <output id="preview" implements="modifiable" on-load="this.set(#qty.value)">1</output>
      </div>

      <ul id="list" implements="renderable">
        <li id="row-1">
          <input class="amount" type="number" value="2.5" step="any" required on-input="#total.set(sum('#list .amount:valid'))">
          <button on-click="#list.render({swap: 'delete', target: '#row-1'}); #total.set(sum('#list .amount:valid'))">×</button>
        </li>
      </ul>
      <button on-click="#list.render({template: #row-tpl, swap: 'beforeend', id: now()}); #total.set(sum('#list .amount:valid'))">Add row</button>
      <template id="row-tpl">
        <li id="row-{id}">
          <input class="amount" type="number" value="3.25" step="any" required on-input="#total.set(sum('#list .amount:valid'))">
          <button on-click="#list.render({swap: 'delete', target: '#row-{id}'}); #total.set(sum('#list .amount:valid'))">×</button>
        </li>
      </template>
      <p class="total">Total:
        <output id="total" implements="modifiable formattable"
                on-load="this.set(sum('#list .amount:valid'))"
                formattable-format="{ style: 'currency', currency: 'USD' }">0</output>
